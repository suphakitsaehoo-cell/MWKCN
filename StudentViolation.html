function renderTable(violations) {

  const container = document.getElementById('contentArea');


  if (!violations || violations.length === 0) {

    container.innerHTML = `

      <div class="empty-state">

        <i class="fa-solid fa-clipboard-check"></i>

        <div class="fw-bold">ไม่พบประวัติการทำผิดกฎ</div>

        <div class="small">นักเรียนคนนี้ยังไม่มีบันทึกความผิดในระบบ</div>

      </div>

    `;

    return;

  }


  let totalCount = 0;


  // สร้างข้อมูลสำหรับทั้งตารางและการ์ด

  let rowsHtml = '';

  let cardsHtml = '';


  violations.forEach((v, idx) => {

    totalCount += parseInt(v.count) || 0;


    let imgCount = 0;

    if (v.img1 && v.img1 !== "-") imgCount++;

    if (v.img2 && v.img2 !== "-") imgCount++;

    if (v.img3 && v.img3 !== "-") imgCount++;


    let evidenceBtn = '';

    if (imgCount > 0) {

      evidenceBtn = `<button class="evidence-btn" onclick="openEvidenceModal(${idx})"><i class="fa-solid fa-image"></i> กดดูหลักฐาน (${imgCount} รูป)</button>`;

    } else {

      evidenceBtn = `<span class="evidence-btn disabled"><i class="fa-solid fa-ban"></i> ไม่มีหลักฐาน</span>`;

    }


    let imgData = JSON.stringify([v.img1, v.img2, v.img3].filter(img => img && img !== "-"));


    // แถวตาราง (Desktop)

    rowsHtml += `

      <tr data-imgs='${imgData}' data-idx="${idx}">

        <td><div class="fw-bold">${v.date}</div></td>

        <td><div class="fw-600">${v.rule}</div>${v.detail !== '-' ? `<div class="small text-muted">${v.detail}</div>` : ''}</td>

        <td class="text-center"><span class="badge bg-light text-dark rounded-pill px-3">${v.count} ครั้ง</span></td>

        <td class="text-center">${evidenceBtn}</td>

      </tr>

    `;


    // การ์ด (Mobile)

    cardsHtml += `

      <div class="violation-card" data-imgs='${imgData}' data-idx="${idx}">

        <div class="card-header-row">

          <div class="card-date"><i class="fa-regular fa-calendar me-1"></i>${v.date}</div>

          <div class="card-count">${v.count} ครั้ง</div>

        </div>

        <div class="card-rule">${v.rule}</div>

        ${v.detail !== '-' ? `<div class="card-detail"><i class="fa-solid fa-circle-info me-1"></i>${v.detail}</div>` : ''}

        <div class="card-evidence">${evidenceBtn}</div>

      </div>

    `;

  });


  container.innerHTML = `

    <div class="d-flex justify-content-between align-items-center mb-3 px-2">

      <div class="small fw-bold text-muted">พบ ${violations.length} รายการ</div>

      ${totalCount >= 5 ? '<div class="danger-badge"><i class="fa-solid fa-triangle-exclamation me-1"></i>สะสมเกิน 5 ครั้ง</div>' : ''}

    </div>


    <!-- Desktop: ตาราง -->

    <div class="data-table-wrapper">

      <table class="data-table">

        <thead>

          <tr>

            <th style="width:18%">วันที่</th>

            <th style="width:42%">รายการความผิด</th>

            <th style="width:15%" class="text-center">จำนวน</th>

            <th style="width:25%" class="text-center">หลักฐาน</th>

          </tr>

        </thead>

        <tbody>

          ${rowsHtml}

        </tbody>

      </table>

    </div>


    <!-- Mobile: การ์ด -->

    <div class="mobile-cards">

      ${cardsHtml}

    </div>

  `;

}


function openEvidenceModal(idx) {

  // หาข้อมูลได้ทั้งจากตาราง (tr) และการ์ด (div)

  let el = document.querySelector(`tr[data-idx="${idx}"]`);

  if (!el) el = document.querySelector(`div[data-idx="${idx}"]`);

  if (!el) return;


  const imgs = JSON.parse(el.getAttribute('data-imgs') || '[]');

  const grid = document.getElementById('modalImgGrid');


  if (imgs.length === 0) return;


  grid.innerHTML = imgs.map(src => `

    <div class="modal-img-item">

      <img src="${src}" alt="หลักฐาน" onclick="window.open('${src}', '_blank')">

    </div>

  `).join('');


  document.getElementById('modalOverlay').classList.add('show');

  setTimeout(() => {

    document.getElementById('evidenceModal').classList.add('show');

  }, 10);

}


function closeModal() {

  document.getElementById('evidenceModal').classList.remove('show');

  setTimeout(() => {

    document.getElementById('modalOverlay').classList.remove('show');

  }, 300);

}


function goBack() {

  if (typeof google !== 'undefined') {

    google.script.run.withSuccessHandler(url => {

      window.location.href = url;

    }).getScriptUrl();

  } else {

    window.close();

  }

}
