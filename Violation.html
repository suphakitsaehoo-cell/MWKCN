function handleSearch(input) {

  const val = input.value.trim().toLowerCase();

  const suggestBox = document.getElementById('suggestBox');

  const clearBtn = document.getElementById('clearSearchBtn');



  clearBtn.style.display = val ? 'flex' : 'none';

  document.getElementById('studentValidation').style.display = 'none';



  if (!val) { suggestBox.style.display = 'none'; return; }



  let matches = [];

  for (let id in nameMasterList) {

    if (id.startsWith(val) || nameMasterList[id].toLowerCase().includes(val)) {

      matches.push({ id, name: nameMasterList[id] });

    }

  }



  if (matches.length > 0) {

    suggestBox.innerHTML = matches.map(m => 

      `<div class="suggest-item" onclick="selectStudent('${m.id}', '${m.name}')">เลขที่ ${m.id} - ${m.name}</div>`

    ).join('');

    suggestBox.style.display = 'block';

  } else {

    suggestBox.style.display = 'none';

  }

}

  document.getElementById('clearSearchBtn').style.display = 'none';

}

function toggleOther(val) {

  const box = document.getElementById('otherBox');

  if (val === "อื่นๆ") {

    box.style.display = "block";

  } else {

    box.style.display = "none";

  }

  document.getElementById('ruleValidation').style.display = 'none';

function handleFiles(files) {

  const fileVal = document.getElementById('fileValidation');

  if (selectedImages.length + files.length > 3) {

    fileVal.style.display = 'block';

    return;

  }

  fileVal.style.display = 'none';



  Array.from(files).forEach(file => {

    if (!file.type.startsWith('image/')) return;

    if (selectedImages.length >= 3) return;



    const reader = new FileReader();

    reader.onload = (e) => {

      selectedImages.push(e.target.result);

      renderPreviews();

    };

    reader.readAsDataURL(file);

  });

}



function renderPreviews() {

  const grid = document.getElementById('previewGrid');

  const dropZone = document.getElementById('dropZone');



  if (selectedImages.length > 0) {

    dropZone.classList.add('has-files');

    dropZone.innerHTML = `<i class="fa-solid fa-check-circle fa-2x mb-2" style="color: var(--ios-green);"></i><div class="fw-bold small">แนบรูปแล้ว ${selectedImages.length}/3</div><div class="small text-muted">แตะเพื่อเพิ่มรูปภาพ</div>`;

  } else {

    dropZone.classList.remove('has-files');

    dropZone.innerHTML = `<i class="fa-solid fa-camera fa-2x mb-2" style="color: var(--ios-blue);"></i><div class="fw-bold small">แตะเพื่อเลือกรูปภาพ</div><div class="small text-muted">รองรับ JPG, PNG (สูงสุด 3 รูป)</div>`;

  }



  grid.innerHTML = selectedImages.map((src, idx) => `

    <div class="preview-item">

      <img src="${src}" alt="preview">

      <button class="preview-remove" onclick="removeImage(${idx})"><i class="fa-solid fa-xmark"></i></button>

    </div>

  `).join('');

}

function showSuccess(msg) {

  const popup = document.getElementById('success-popup');

  document.getElementById('success-msg').innerText = msg;

  popup.animate([

    { opacity: 1, transform: 'translate(-50%, -50%) scale(1)' }

  ], { duration: 400, easing: 'cubic-bezier(0.175, 0.885, 0.32, 1.275)', fill: 'forwards' });



  setTimeout(() => {

    popup.animate([

      { opacity: 1, transform: 'translate(-50%, -50%) scale(1)' },

      { opacity: 0, transform: 'translate(-50%, -50%) scale(0.8)' }

    ], { duration: 300, fill: 'forwards' }).onfinish = () => {

      popup.style.display = 'none';

    };

  }, 2500);

}

  content.innerHTML = '<div class="report-empty"><i class="fa-solid fa-spinner fa-spin fa-2x"></i><br>กำลังดึงข้อมูลจากระบบ...</div>';



  if (typeof google !== 'undefined') {

    google.script.run

      .withSuccessHandler((data) => {

        btn.disabled = false;

        btn.innerHTML = '<i class="fa-solid fa-magnifying-glass me-1"></i> ดึงข้อมูล';

        if (data && data.error) {

          content.innerHTML = '<div class="report-empty"><i class="fa-solid fa-triangle-exclamation fa-2x mb-2" style="opacity:0.4; color:var(--ios-red);"></i><br>' + data.error + '</div>';

          document.getElementById('btnExportPdf').style.display = 'none';

          return;

        }

        renderReport(data, month, year);

      })

      .withFailureHandler((err) => {

        btn.disabled = false;

        btn.innerHTML = '<i class="fa-solid fa-magnifying-glass me-1"></i> ดึงข้อมูล';

        content.innerHTML = '<div class="report-empty"><i class="fa-solid fa-triangle-exclamation fa-2x mb-2" style="opacity:0.4; color:var(--ios-red);"></i><br>เกิดข้อผิดพลาด: ' + err + '</div>';

      })

      .getMonthlySummary(month, year);

  } else {

    btn.disabled = false;

    btn.innerHTML = '<i class="fa-solid fa-magnifying-glass me-1"></i> ดึงข้อมูล';

    content.innerHTML = '<div class="report-empty"><i class="fa-solid fa-wifi-slash fa-2x mb-2" style="opacity:0.4; color:var(--ios-red);"></i><br>ไม่พบการเชื่อมต่อกับระบบหลังบ้าน (Google Apps Script)</div>';

    document.getElementById('btnExportPdf').style.display = 'none';

  }

}



function renderReport(data, month, year) {

  const content = document.getElementById('reportContent');

  const title = 'เดือน' + data.month + ' พ.ศ. ' + data.yearBE;


  // แปลงค่าเป็นตัวเลขให้แน่นอน ป้องกันกรณี GAS ส่งค่ากลับมาเป็นข้อความ

  const violationTotalFine = Number(data.violationTotalFine) || 0;

  const roomTotal = Number(data.roomTotal) || 0;

  const expenseTotal = Number(data.expenseTotal) || 0;

  // กำหนดภายหลังจากคำนวณ unpaidTotal แล้ว

  const cumulativeTotal = Number(data.cumulativeTotal) || 0;


  const unpaidRoom = data.unpaidRoomList || [];

  const unpaidFine = data.unpaidFineList || [];

  const unpaidRoomTotal = Number(data.unpaidRoomTotal) || 0;

  const unpaidFineTotal = Number(data.unpaidFineTotal) || 0;


  // ใช้ data.unpaidTotal เป็นหลัก หากไม่มี ให้คำนวณจากยอดค้างค่าห้องและค่าปรับ

  const unpaidTotal = Number.isFinite(Number(data.unpaidTotal))

    ? Number(data.unpaidTotal)

    : unpaidRoomTotal + unpaidFineTotal;


  // ดุลสุทธิประจำเดือน = ทำผิดกฎ + ค่าห้อง + เลยกำหนด - รายจ่าย

  const netBalance = violationTotalFine + roomTotal + unpaidTotal - expenseTotal;



  document.getElementById('btnExportPdf').style.display = 'inline-flex';


  // 1) Summary Block: ยอดสะสมด้านบน + การ์ดสรุป 4 รายการตรงกลาง

  const cardsHtml = `

    <div class="summary-overview" style="display:block; margin-bottom:12px;">

      <div style="background:linear-gradient(135deg, rgba(52,199,89,0.1) 0%, rgba(0,122,255,0.06) 100%); border-radius:20px; padding:16px; border:1px solid rgba(52,199,89,0.25); text-align:center;">

        <div style="font-size:0.8rem; font-weight:700; color:#8e8e93; text-transform:uppercase; letter-spacing:0.5px;">ยอดเงินสุทธิสะสมทั้งหมด</div>

        <div style="font-size:1.5rem; font-weight:800; color:#34c759; margin-top:4px;">฿${fmtMoney(cumulativeTotal)}</div>

      </div>

    </div>

    <div class="summary-cards">

      <div class="summary-card card-red">

        <div class="card-label">ยอดปรับเงินทำผิดกฎ</div>

        <div class="card-value">฿${fmtMoney(violationTotalFine)}</div>

      </div>

      <div class="summary-card card-green">

        <div class="card-label">ยอดเก็บเงินค่าห้อง</div>

        <div class="card-value">฿${fmtMoney(roomTotal)}</div>

      </div>

      <div class="summary-card card-red">

        <div class="card-label">ชำระล่าช้า/เลยกำหนด</div>

        <div class="card-value">฿${fmtMoney(unpaidTotal)}</div>

      </div>

      <div class="summary-card card-orange">

        <div class="card-label">ยอดรายจ่ายทั้งสิ้น</div>

        <div class="card-value">฿${fmtMoney(expenseTotal)}</div>

      </div>

    </div>

  `;


  // 2) ตารางรายการทำผิดกฎและค่าปรับรายบุคคล

  let vHtml = `<div class="report-section-title"><i class="fa-solid fa-user-slash" style="color:var(--ios-red);"></i>รายการทำผิดกฎและค่าปรับรายบุคคล</div>`;

  const viols = data.violations || [];

  const persons = data.violationsSummary || [];

  if (viols.length === 0 && persons.length === 0) {

    vHtml += '<div class="report-empty">ไม่พบรายการทำผิดกฎในช่วงเวลาที่เลือก</div>';

  } else {

    vHtml += `

      <table class="report-table tbl-5cols">

        <thead><tr><th>เลขที่</th><th>ชื่อ-นามสกุล</th><th>รายการทำผิด</th><th class="num-cell">จำนวนครั้ง</th><th class="num-cell">ค่าปรับ (บาท)</th></tr></thead>

        <tbody>

          ${viols.map(v => `

            <tr><td>${v.id}</td><td>${v.name || '-'}</td><td>${v.rule || '-'}</td><td class="num-cell">${v.count}</td><td class="num-cell">${fmtMoney(v.fine)}</td></tr>`).join('')}

          <tr class="total-row"><td colspan="4">ยอดปรับเงินทั้งหมดประจำเดือน</td><td class="num-cell">฿${fmtMoney(violationTotalFine)}</td></tr>

        </tbody>

      </table>`;

    if (persons.length > 0) {

      vHtml += `

        <div class="report-section-title" style="margin-top:8px;"><i class="fa-solid fa-users" style="color:var(--ios-blue);"></i>สรุปสะสมรายการทำผิดกฎ</div>

        <table class="report-table tbl-4cols">

          <thead><tr><th>เลขที่</th><th>ชื่อ-นามสกุล</th><th>จำนวนครั้งสะสม</th><th class="num-cell">ค่าปรับสะสม (บาท)</th></tr></thead>

          <tbody>${persons.map(p => `<tr><td>${p.id}</td><td>${p.name || '-'}</td><td class="num-cell">${p.totalTimes}</td><td class="num-cell">${fmtMoney(p.totalFines)}</td></tr>`).join('')}</tbody>

        </table>`;

    }

  }


  // 3) ตารางสรุปการจัดเก็บค่าห้องรายรอบ

  let rHtml = `<div class="report-section-title"><i class="fa-solid fa-house-circle-check" style="color:var(--ios-green);"></i>การจัดเก็บค่าห้องรายรอบของการเก็บเงิน (${title})</div>`;

  const rounds = data.roomCollectionRounds || [];

  if (rounds.length === 0) {

    rHtml += '<div class="report-empty">ไม่พบรอบเก็บเงินค่าห้องในช่วงเวลาที่เลือก</div>';

  } else {

    rHtml += `

      <div class="table-scroll"><table class="report-table tbl-3cols">

        <thead><tr><th>รอบเก็บเงิน (วันที่)</th><th class="num-cell">จำนวนผู้ชำระ</th><th class="num-cell">ยอดรวม (บาท)</th></tr></thead>

        <tbody>

          ${rounds.map(r => `<tr><td>${r.label}</td><td class="num-cell">${r.paidCount} คน</td><td class="num-cell">${fmtMoney(r.total)}</td></tr>`).join('')}

          <tr class="total-row"><td colspan="2">ยอดเก็บเงินค่าห้องทั้งห้องประจำเดือน</td><td class="num-cell">฿${fmtMoney(roomTotal)}</td></tr>

        </tbody>

      </table></div>`;

  }


  // 4) ตารางผู้ค้างชำระ/เลยกำหนด — วางต่อท้าย rHtml ตามลำดับที่ต้องการ

  let uHtml = '';

  if (unpaidRoom.length > 0 || unpaidFine.length > 0) {

    uHtml += `<div class="unpaid-section"><div class="report-section-title" style="color:var(--ios-red);"><i class="fa-solid fa-bell"></i>รายการชำระล่าช้า/เลยกำหนด (ชำระแล้ว)<span class="unpaid-badge">ยอดชำระล่าช้ารวม ฿${fmtMoney(unpaidTotal)}</span></div>`;

    if (unpaidRoom.length > 0) {

      uHtml += `<div style="font-size:0.85rem; font-weight:700; color:var(--ios-red); margin:8px 0 6px;"><i class="fa-solid fa-house-circle-exclamation me-1"></i>ผู้ค้างชำระค่าห้อง</div>

        ${unpaidRoom.map(u => `<tr><td>${u.id}</td><td>${u.name || '-'}</td><td class="num-cell">฿${fmtMoney(u.amount)}</td></tr>`).join('')}

        <tr class="total-row"><td colspan="2">ยอดค้างค่าห้องรวม</td><td class="num-cell">฿${fmtMoney(unpaidRoomTotal)}</td></tr></tbody></table></div>`;

    }

    if (unpaidFine.length > 0) {

      uHtml += `<div style="font-size:0.85rem; font-weight:700; color:var(--ios-red); margin:8px 0 6px;"><i class="fa-solid fa-gavel me-1"></i>ผู้ค้างชำระค่าปรับ</div>

        <div class="table-scroll"><table class="report-table tbl-3cols"><thead><tr><th>เลขที่</th><th>ชื่อ-นามสกุล</th><th class="num-cell">ยอดค้างชำระ (บาท)</th></tr></thead><tbody>

        ${unpaidFine.map(u => `<tr><td>${u.id}</td><td>${u.name || '-'}</td><td class="num-cell">฿${fmtMoney(u.amount)}</td></tr>`).join('')}

        <tr class="total-row"><td colspan="2">ยอดค้างค่าปรับรวม</td><td class="num-cell">฿${fmtMoney(unpaidFineTotal)}</td></tr></tbody></table></div>`;

    }

    uHtml += '</div>';

  } else {

    uHtml = `<div class="unpaid-section" style="border-color:rgba(52,199,89,0.3); background:linear-gradient(135deg, rgba(52,199,89,0.06) 0%, rgba(52,199,89,0.02) 100%);"><div style="text-align:center; color:#34c759; font-weight:700;"><i class="fa-solid fa-circle-check me-1"></i>ไม่พบผู้ค้างชำระค่าห้องและค่าปรับทั้งหมด</div></div>`;

  let eHtml = `<div class="report-section-title"><i class="fa-solid fa-money-bill-transfer" style="color:var(--ios-orange);"></i>รายการบันทึกรายจ่ายประจำเดือน (${title})</div>`;

  const exps = data.expenses || [];

  if (exps.length === 0) {

    eHtml += '<div class="report-empty">ไม่พบรายการรายจ่ายในช่วงเวลาที่เลือก</div>';

  // 6) ป๊อบอัพดุลสุทธิประจำเดือนด้านล่างสุด

  const netHtml = `<div style="background:linear-gradient(135deg, rgba(0,122,255,0.08) 0%, rgba(0,122,255,0.02) 100%); border-radius:20px; padding:18px; text-align:center; border:1px solid rgba(0,122,255,0.15); margin-bottom:8px;">

    <div style="font-size:0.8rem; font-weight:700; color:#8e8e93; text-transform:uppercase; letter-spacing:0.5px;">ดุลสุทธิประจำเดือน (ยอดปรับเงินทำผิดกฎ + ค่าห้อง + ชำระล่าช้า - รายจ่าย)</div>

    <div style="font-size:1.6rem; font-weight:800; color:var(--ios-blue); margin-top:6px;">฿${fmtMoney(netBalance)}</div>

    <div style="font-size:0.78rem; color:#8e8e93; margin-top:8px;">฿${fmtMoney(violationTotalFine)} + ฿${fmtMoney(roomTotal)} + ฿${fmtMoney(unpaidTotal)} - ฿${fmtMoney(expenseTotal)}</div>

  </div>`;


  // ลำดับ: ยอดสะสมด้านบน → การ์ด 4 รายการ → ตารางต่าง ๆ → ดุลสุทธิด้านล่าง

  content.innerHTML = cardsHtml + netHtml + vHtml + rHtml + uHtml + eHtml;

}

function exportReportPdf() {

  const btn = document.getElementById('btnExportPdf');

  if (btn) {

    btn.disabled = true;

    btn.innerHTML = '<i class="fa-solid fa-spinner fa-spin me-1"></i> กำลังเตรียม PDF...';

  }

  document.body.classList.add('printing');

  // ชื่อไฟล์แนะนำ (เบราว์เซอร์อาจใช้เป็นชื่อไฟล์อัตโนมัติ)

  const monthStr = document.getElementById('reportMonth').value;

  const yearStr = document.getElementById('reportYear').value.trim();

  const labelMonth = ['ม.ค.','ก.พ.','มี.ค.','เม.ย.','พ.ค.','มิ.ย.','ก.ค.','ส.ค.','ก.ย.','ต.ค.','พ.ย.','ธ.ค.'][parseInt(monthStr, 10) - 1] || '';

  document.title = 'สรุปรายงานประจำเดือน_' + (yearStr ? labelMonth + '_' + yearStr : 'สรุปรายงาน');


  setTimeout(function() {

    document.body.classList.remove('printing');

    document.title = 'ระบบบันทึกการทำผิดกฎ - MWKCN';

    if (btn) {

      btn.disabled = false;

      btn.innerHTML = '<i class="fa-solid fa-file-pdf me-1"></i> บันทึกเป็น PDF';

    }

  }, 300);

}
